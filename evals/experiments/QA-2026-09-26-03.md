```python
"""E7's measure on two critic samples: six greedy verdict-based questions a case.

Also counts the facts with no place in the model (question and subject
spellings) and how many of those keys both samples share.
"""

import json
import sys
from pathlib import Path

from analysis_service.claims import UnknownRef
from analysis_service.questions import _greedy


def waiting(path):
    claims = json.loads(Path(path).read_text())["claims"]
    out = {}
    for claim in claims:
        verdict = claim["verdict"]
        refs = [
            UnknownRef.model_validate(r) for r in verdict.get("related_unknowns", [])
        ]
        if refs:
            out[claim["id"]] = refs
    return out


def settles(questions, facts):
    return sum(1 for keys in facts.values() if keys <= set(questions))


def main(a_dir, b_dir, cases):
    shared = total = own_a = own_b = cross_a = cross_b = 0
    place_a = place_b = place_both = typed = free = 0
    for case in cases:
        wa = waiting(f"{a_dir}/{case}.rulings.json")
        wb = waiting(f"{b_dir}/{case}.rulings.json")
        fa = {c: {r.key for r in refs} for c, refs in wa.items()}
        fb = {c: {r.key for r in refs} for c, refs in wb.items()}
        qa, qb = _greedy(fa)[:6], _greedy(fb)[:6]
        shared += len(set(qa) & set(qb))
        total += len(qa)
        own_a += settles(qa, fa)
        cross_a += settles(qb, fa)
        own_b += settles(qb, fb)
        cross_b += settles(qa, fb)
        na = {
            r.key
            for refs in wa.values()
            for r in refs
            if r.spellings in (("question",), ("subject",))
        }
        nb = {
            r.key
            for refs in wb.values()
            for r in refs
            if r.spellings in (("question",), ("subject",))
        }
        place_a += len(na)
        place_b += len(nb)
        place_both += len(na & nb)
        for refs in (*wa.values(), *wb.values()):
            typed += sum(1 for r in refs if r.spellings == ("question",))
            free += sum(1 for r in refs if r.spellings == ("subject",))
    print(f"cases {len(cases)}: top-6 lists share {shared} of {total}")
    print(f"A's list settles {own_a} of A; B's list settles {cross_a} of A")
    print(f"B's list settles {own_b} of B; A's list settles {cross_b} of B")
    print(f"no-place facts: A {place_a}, B {place_b}, in both {place_both}")
    rate = free / (typed + free) if typed + free else float("nan")
    print(f"citations: typed {typed}, free text {free}, fallback {rate:.0%}")


if __name__ == "__main__":
    main(sys.argv[1], sys.argv[2], sys.argv[3].split(","))
```

## E13: an early list settles about 60% of the report list

An early list sees only the System Model and STRIDE's candidates. It ranks
each element's attributes and question kinds by how often other cases cite
them per element of that type, times one plus the number of candidates that
name the element. On E11's two terra passes:

| Questions a case | Report-ranked (a / b) | Early (a / b) | Early share |
|---:|---:|---:|---:|
| 3 | 37 / 36 | 15 / 15 | 41% / 42% |
| 6 | 58 / 61 | 37 / 36 | 64% / 59% |
| 10 | 93 / 92 | 59 / 57 | 63% / 62% |

The type prior alone reaches 41% and 44% at six. A boost for a candidate's
`unknown` attribute helps the first three questions (62% and 75%) and hurts
after that. Kinds are 261 of the 577 citations and zones 16, so an early list
now reaches most of the facts that E5 said only the critic raised.

The count reads findings that the lanes wrote without answers. It cannot see
an early answer change what the lanes write, which is the main reason to ask
early.

```python
"""How much an early question list settles, against the list after the report.

Findings: each needs-info verdict's open facts in one critic pass. The early
list sees only the case's System Model. It ranks every (element, attribute)
and (element, kind) pair by a prior read from the other 12 cases: how often a
finding cites that attribute or kind, per element of that type. Leave one
case out, so a case never ranks itself.
"""

import json
import sys
from collections import Counter
from pathlib import Path

from analysis_service.candidates import generate_candidates
from analysis_service.claims import UnknownRef
from analysis_service.frameworks import PACKAGES
from analysis_service.question_kinds import QUESTION_KINDS
from analysis_service.questions import _greedy
from analysis_service.system_model import SystemModel

VARIANT = sys.argv[2]

RUNS = Path(sys.argv[1])
REPORTS = Path(
    "evals/baselines/6bff717-gpt-5.6-terra-24dda4db/mstarks01-e32ad9c5.reports"
)
COUNTS = [3, 6, 10]


def kind_of(element_id):
    return element_id.split(":", 1)[0]


def field(key):
    element_id, attribute, assertion, subject, question = key
    return attribute or question


def findings(case):
    claims = json.loads((RUNS / f"{case}.rulings.json").read_text())["claims"]
    out = {}
    for claim in claims:
        refs = [
            UnknownRef.model_validate(r)
            for r in claim["verdict"].get("related_unknowns", [])
        ]
        if refs:
            out[claim["id"]] = {r.key for r in refs}
    return out


def elements(case):
    report = json.loads((REPORTS / f"{case}.report.json").read_text())
    model = SystemModel.model_validate(report["system_model"])
    ids = [e.id for e in model.elements()]
    stride = PACKAGES["stride"]
    named, leads = Counter(), Counter()
    for found in generate_candidates(model, stride.lanes, stride.rules).values():
        for candidate in found.candidates:
            named.update(set(candidate.element_ids))
            for name, value in candidate.facts.items():
                if value == "unknown":
                    leads[(candidate.element_ids[0], name)] += 1
    return ids, named, leads


def settles(questions, facts):
    asked = set(questions)
    return sum(1 for keys in facts.values() if keys <= asked)


cases = sorted(
    p.name.removesuffix(".rulings.json") for p in RUNS.glob("*.rulings.json")
)
data = {case: (findings(case), elements(case)) for case in cases}
totals = Counter()
reachable = unreachable = 0
for case in cases:
    facts, (ids, named, leads) = data[case]
    cited, seen = Counter(), Counter()
    for other in cases:
        if other == case:
            continue
        other_facts, (other_ids, _, _) = data[other]
        seen.update(kind_of(i) for i in other_ids)
        for keys in other_facts.values():
            for key in keys:
                if key[0] and field(key):
                    cited[(kind_of(key[0]), field(key))] += 1
    prior = {pair: n / seen[pair[0]] for pair, n in cited.items() if seen[pair[0]]}
    candidates = []
    for element_id in ids:
        for (etype, name), score in prior.items():
            if etype != kind_of(element_id):
                continue
            is_kind = name in QUESTION_KINDS
            key = (
                (element_id, "", "", "", name)
                if is_kind
                else (element_id, name, "", "", "")
            )
            if VARIANT in ("named", "leads"):
                score *= 1 + named[element_id]
            if VARIANT == "leads" and leads[(element_id, name)]:
                score += 100 * leads[(element_id, name)]
            candidates.append((score, key))
    candidates.sort(key=lambda c: (-c[0], c[1]))
    early = [key for _, key in candidates]
    report = _greedy(facts)
    totals["findings"] += len(facts)
    for n in COUNTS:
        totals[f"report {n}"] += settles(report[:n], facts)
        totals[f"early {n}"] += settles(early[:n], facts)
    totals["report all"] += settles(report, facts)
    totals["early all"] += settles(early, facts)
    totals["early questions"] += len(early)
    for keys in facts.values():
        for key in keys:
            if key[0] and field(key):
                reachable += 1
            else:
                unreachable += 1
print(f"{len(cases)} cases, {totals['findings']} needs-info findings")
for n in COUNTS:
    r, e = totals[f"report {n}"], totals[f"early {n}"]
    print(
        f"{n} a case: report-ranked settle {r}, early settle {e} ({e / r:.0%} of report)"
    )
print(
    f"all: report {totals['report all']}, early {totals['early all']} with {totals['early questions']} early questions"
)
print(
    f"citations an early question can ask: {reachable}; cannot (subject, assertion): {unreachable}"
)
```

```python
"""E13 again, with the production early_questions as the ranking (#1289)."""

import json
import sys
from collections import Counter
from pathlib import Path

from analysis_service.candidates import generate_candidates
from analysis_service.claims import UnknownRef
from analysis_service.early_questions import PriorRow, early_questions
from analysis_service.frameworks import PACKAGES
from analysis_service.question_kinds import QUESTION_KINDS
from analysis_service.questions import _greedy
from analysis_service.system_model import SystemModel
from evals.harness.question_prior import count, replay_findings

REPORTS = Path(
    "evals/baselines/6bff717-gpt-5.6-terra-24dda4db/mstarks01-e32ad9c5.reports"
)
PASSES = Path("evals/runs/20260927T-critic-question-kinds")
COUNTS = [3, 6, 10]


def kind_of(element_id):
    return element_id.split(":", 1)[0]


def field(key):
    return key[1] or key[4]


def covered(questions, facts):
    asked = set(questions)
    return sum(1 for keys in facts.values() if keys <= asked)


def e13_ranking(model, facts_by_case, case, models):
    """The published E13 'named' ranking, copied from QA-2026-09-26-03.md."""
    cited, seen = Counter(), Counter()
    for other, other_facts in facts_by_case.items():
        if other == case:
            continue
        seen.update(kind_of(e.id) for e in models[other].elements())
        for keys in other_facts.values():
            for key in keys:
                if key[0] and field(key):
                    cited[(kind_of(key[0]), field(key))] += 1
    prior = {p: n / seen[p[0]] for p, n in cited.items() if seen[p[0]]}
    stride = PACKAGES["stride"]
    named = Counter()
    for found in generate_candidates(model, stride.lanes, stride.rules).values():
        for candidate in found.candidates:
            named.update(set(candidate.element_ids))
    ranked = []
    for element in model.elements():
        for (etype, name), score in prior.items():
            if etype != kind_of(element.id):
                continue
            key = (
                (element.id, "", "", "", name)
                if name in QUESTION_KINDS
                else (element.id, name, "", "", "")
            )
            ranked.append((score * (1 + named[element.id]), key))
    ranked.sort(key=lambda c: (-c[0], c[1]))
    return [key for _, key in ranked]


def run_pass(name):
    runs = PASSES / name
    cases = sorted(
        p.name.removesuffix(".rulings.json") for p in runs.glob("*.rulings.json")
    )
    models, findings = {}, {}
    for case in cases:
        report = json.loads((REPORTS / f"{case}.report.json").read_text())
        models[case] = SystemModel.model_validate(report["system_model"])
        findings[case] = replay_findings(runs / f"{case}.rulings.json")
    facts_by_case = {
        case: {i: {ref.key for ref in refs} for i, refs in enumerate(findings[case])}
        for case in cases
    }
    totals = Counter()
    lengths = []
    for case in cases:
        facts = facts_by_case[case]
        rates = count(
            (other, models[other], findings[other]) for other in cases if other != case
        )
        prior = {
            "stride": PriorRow(runs=("loo",), revision=None, cases=12, rates=rates)
        }
        rankings = {
            "report": _greedy(facts),
            "e13": e13_ranking(models[case], facts_by_case, case, models),
            "production": [
                q.key for q in early_questions(models[case], ["stride"], None, prior)
            ],
        }
        lengths.append(len(rankings["production"]))
        totals["findings"] += len(facts)
        for label, ranked in rankings.items():
            totals[f"{label} len"] += len(ranked)
            totals[f"{label} all"] += covered(ranked, facts)
            for n in COUNTS:
                totals[f"{label} {n}"] += covered(ranked[:n], facts)
    return cases, totals, lengths


for name in sys.argv[1:] or ["a", "b"]:
    cases, totals, lengths = run_pass(name)
    print(f"pass {name}: {len(cases)} cases, {totals['findings']} needs-info findings")
    for n in [*COUNTS, "all"]:
        r = totals[f"report {n}"]
        row = [f"report {r}"]
        for label in ("e13", "production"):
            v = totals[f"{label} {n}"]
            row.append(f"{label} {v} ({v / r:.0%})" if r else f"{label} {v}")
        print(f"  {n}: " + ", ".join(row))
    print(
        f"  list length: e13 {totals['e13 len']}, production {totals['production len']}"
        f" (per case {min(lengths)}-{max(lengths)})"
    )
```

These are four decisions on #1289, measured offline on production code before

**D4, the order.** The shipped order now asks first the question that
completes the most findings. Where no single question completes one, it asks
the facts of the finding with the fewest left. Findings covered at 3, 6 and 10

|---|---:|---:|

The gain is small on the grounds, which are the first section of a report's

**D1, eligibility.** If every open attribute and zone that the prior does not

and 155 where it covered 155.

**D2 and D3.** Over 248 archived reports, 191 of 3,161 attribute questions

section. On the 13 Baseline reports, 13 of the 156 findings that the count

```python
"""E17: offline evidence for the #1289 design decisions, on production code."""

import json
from collections import Counter
from pathlib import Path

from analysis_service.early_questions import (
    PriorRow,
    _open_key,
    early_questions,
    element_type,
)
from analysis_service.questions import _greedy
from analysis_service.report import Report
from analysis_service.system_model import ZONE_ATTRIBUTE, SystemModel, attribute_names
from evals.harness.question_prior import count, replay_findings

REPORTS = Path(
    "evals/baselines/6bff717-gpt-5.6-terra-24dda4db/mstarks01-e32ad9c5.reports"
)
PASSES = Path("evals/runs/20260927T-critic-question-kinds")
COUNTS = (3, 6, 10)


def most_cited_first(open_facts):
    """The order before E17: the fact the most still-open findings cite."""
    left = {f: set(keys) for f, keys in open_facts.items() if keys}
    order = []
    while left:
        cites = Counter(k for keys in left.values() for k in keys)
        key = min(cites.items(), key=lambda item: (-item[1], item[0]))[0]
        order.append(key)
        left = {f: keys - {key} for f, keys in left.items() if keys - {key}}
    return order


def covered(questions, facts):
    asked = set(questions)
    return sum(1 for keys in facts.values() if keys <= asked)


def compare(label, per_case):
    t = Counter()
    for facts in per_case:
        t["findings"] += len(facts)
        for n in COUNTS:
            t[f"old {n}"] += covered(most_cited_first(facts)[:n], facts)
            t[f"new {n}"] += covered(_greedy(facts)[:n], facts)
    print(f"D4 {label}: {t['findings']} findings")
    for n in COUNTS:
        print(
            f"  at {n}: most cited first {t[f'old {n}']}, completes most first {t[f'new {n}']}"
        )


grounds = []
for path in sorted(REPORTS.glob("*.report.json")):
    report = Report.model_validate_json(path.read_text())
    evidence = {}
    for block in report.analyses:
        for claim in block.all_claims():
            keys = {ref.key for ref in claim.unknown_grounds()}
            if keys:
                evidence[(block.framework, claim.id)] = keys
    grounds.append(evidence)
compare("grounds of the 13 Baseline reports", grounds)

for name in ("a", "b"):
    runs = PASSES / name
    cases = sorted(
        p.name.removesuffix(".rulings.json") for p in runs.glob("*.rulings.json")
    )
    models = {
        c: SystemModel.model_validate(
            json.loads((REPORTS / f"{c}.report.json").read_text())["system_model"]
        )
        for c in cases
    }
    findings = {c: replay_findings(runs / f"{c}.rulings.json") for c in cases}
    facts = [
        {i: {r.key for r in refs} for i, refs in enumerate(findings[c])} for c in cases
    ]
    compare(f"critic pass {name}", facts)
    t = Counter()
    for case, case_facts in zip(cases, facts):
        model = models[case]
        rates = count((o, models[o], findings[o]) for o in cases if o != case)
        prior = {"stride": PriorRow(("loo",), None, 12, rates)}
        shipped = [q.key for q in early_questions(model, ["stride"], None, prior)]
        extra = {
            key
            for element in model.elements()
            for field in (*attribute_names(element), ZONE_ATTRIBUTE)
            if hasattr(element, field)
            and field not in rates.get(element_type(element), {})
            and (key := _open_key(model, element, field)) is not None
        } - set(shipped)
        t["shipped"] += len(shipped)
        t["widened"] += len(shipped) + len(extra)
        t["covered"] += covered(shipped, case_facts)
        t["covered widened"] += covered([*shipped, *sorted(extra)], case_facts)
    print(
        f"D1 pass {name}: list {t['shipped']} -> {t['widened']},"
        f" covered in all {t['covered']} -> {t['covered widened']}"
    )
```

## E20: how rounds of early questions move, replayed offline

`run.py early-rounds` answers the early questions of 248 archived extracted
models round after round: 217 STRIDE and 31 ASVS. Each round shows at most 10
questions with a score of 1 or more. The service's own question set, admission

**The blessed model answers too little to test drift.** Only 58 of 2,819
