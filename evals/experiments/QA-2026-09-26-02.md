## Reproduction

Run from the repository root with `uv run python -`, and paste this script:

```python
from collections import Counter
from pathlib import Path

from analysis_service.open_facts import open_facts
from analysis_service.report import Report

CASES = (
    "02-iot-fleet-telemetry",
    "05-cookbook-queue-webapp",
    "10-cookbook-generic-cms",
)
ARMS = {
    "stage1": (
        "evals/runs/20260926T-sweep-b/b-cases02-08.reports/{0}.report.json",
        "evals/runs/20260926T-sweep-b/c-cases09-15.reports/{0}.report.json",
    ),
    "closing": ("evals/runs/20260926T-adr0030-closing/{0}.reports/{0}.report.json",),
}
CAPS = (3, 6, 10)


def settled(needs, cap):
    """How many findings the ``cap`` most-cited questions settle, asked greedily."""
    left = {claim_id: set(keys) for claim_id, keys in needs.items()}
    done = 0
    for _ in range(cap):
        if not left:
            break
        cited = Counter(key for keys in left.values() for key in keys)
        best = cited.most_common(1)[0][0]
        for claim_id in list(left):
            left[claim_id].discard(best)
            if not left[claim_id]:
                del left[claim_id]
                done += 1
    return done


def main():
    for arm, patterns in ARMS.items():
        total = Counter()
        for case in CASES:
            path = next(
                Path(pattern.format(case))
                for pattern in patterns
                if Path(pattern.format(case)).exists()
            )
            report = Report.model_validate_json(path.read_text(encoding="utf-8"))
            block = next(b for b in report.analyses if b.framework == "stride")
            needs = {
                claim.id: {ref.key for ref in claim.verdict.related_unknowns}
                for claim in block.claims
                if claim.verdict.status == "needs-info"
            }
            statuses = Counter(claim.verdict.status for claim in block.claims)
            facts = len(open_facts(block, report.system_model))
            print(f"{arm:8} {case[:2]} {dict(statuses)} open facts {facts}")
            total.update(statuses)
            total["open facts"] += facts
            for cap in CAPS:
                total[f"settled by {cap}"] += settled(needs, cap)
        print(f"{arm:8} all {dict(total)}")


if __name__ == "__main__":
    main()
```
