# QA-2026-10-13-07: a cited sentence that code checks, quoted from memory

QA-2026-10-13-06 found that luna claims "cannot be disabled" for traps. This
audit asked whether requiring a verifiable citation removes those claims. The
model gives the documentation URL and the sentence on it that settles its
answer; code fetches the URL from an allowlist of documentation hosts (https
only, every redirect re-checked, no private addresses, 3 MB and 15 s limits)
and checks the sentence with the validity gate's own matcher
(`grounding.verify_quote`). An unverified claim counts as no claim. The
application is not touched; the harness is restated below.

## One pass (E1)

`gpt-5.6-luna` on OpenAI, the same 39-item key, one pass, $0.019 of a $0.10
grant (`QA-2026-10-13-07/luna-answers-verified.json`):

| Answers | Count |
| --- | --- |
| True property claimed, sentence not on the page | 17 |
| True property claimed, page returned 404 | 3 |
| Trap claimed "cannot be disabled", sentence not on the page | 2 |
| Trap answered "can be disabled" | 11 |
| Cited a host off the allowlist | 6 |

No sentence verified. The sentences are close paraphrases: for Cloud Storage
all six content words appear on the page, but not the sentence the model wrote
("Cloud Storage always encrypts your data on the server side before it is
written to disk"). Quoting from memory produces text that cannot be checked,
so the check refuses true claims as surely as false ones. The remaining four
passes were not run; they could only confirm a recall of zero.

The design that could work is a different one: code fetches the page first,
and the model copies a sentence from the text it is given.

## The harness

```python
"""The trap test again, where a "cannot be disabled" claim counts only if code verifies it.

The model cites a documentation URL and the sentence on it that supports its
answer. Code fetches the URL from an allowlist of documentation hosts and
checks the sentence with the validity gate's own matcher. A claim whose page
cannot be fetched, or whose sentence is not on it, is not a claim: the
question is asked. Nothing here touches the application.
"""
import ipaddress, json, os, socket, sys
from html.parser import HTMLParser
from pathlib import Path
from urllib.error import HTTPError, URLError
from urllib.parse import urljoin, urlparse
from urllib.request import HTTPRedirectHandler, Request, build_opener

from analysis_service.grounding import verify_quote
from analysis_service.model_gate import completion
from analysis_service.vendors import VENDORS, CredentialMode

sys.path.insert(0, str(Path(__file__).parent))
from trap_test import ITEMS  # noqa: E402  the same verified key

ALLOWED_HOSTS = {
    "docs.aws.amazon.com", "aws.amazon.com", "learn.microsoft.com", "cloud.google.com",
    "docs.cloud.google.com", "docs.stripe.com", "stripe.com", "docs.snowflake.com",
    "www.mongodb.com", "auth0.com", "datatracker.ietf.org", "www.rfc-editor.org",
    "rfc-editor.org", "www.wireguard.com", "openid.net", "www.w3.org",
}
MAX_BYTES = 3_000_000
TIMEOUT = 15
MAX_HOPS = 4

PROMPT = """For each item, answer from the vendor's or the standard's own documentation.
"on_by_default": is the security property active without the customer doing anything?
"can_disable": can a customer turn the property off, choose an unprotected option, or create the resource without it?
Answer "can_disable": true whenever any supported configuration lacks the property.
"source": the https URL of the documentation page that settles "can_disable".
"quote": one sentence copied exactly from that page that settles "can_disable". Copy it verbatim; do not paraphrase.
Return only JSON: {"answers": [{"id": "...", "on_by_default": true|false, "can_disable": true|false, "source": "...", "quote": "..."}]}

Items:
"""


class _NoRedirect(HTTPRedirectHandler):
    def redirect_request(self, *args, **kwargs):
        return None


class _Text(HTMLParser):
    """The visible text of a page; script and style are dropped."""

    def __init__(self):
        super().__init__()
        self.parts, self._skip = [], 0

    def handle_starttag(self, tag, attrs):
        if tag in ("script", "style", "noscript"):
            self._skip += 1

    def handle_endtag(self, tag):
        if tag in ("script", "style", "noscript") and self._skip:
            self._skip -= 1

    def handle_data(self, data):
        if not self._skip:
            self.parts.append(data)


def _refused(url):
    """Why this URL may not be fetched, or None."""
    parsed = urlparse(url)
    if parsed.scheme != "https":
        return "not https"
    if parsed.hostname not in ALLOWED_HOSTS:
        return f"host {parsed.hostname!r} not on the allowlist"
    try:
        addresses = {info[4][0] for info in socket.getaddrinfo(parsed.hostname, 443)}
    except socket.gaierror:
        return "host does not resolve"
    for address in addresses:
        ip = ipaddress.ip_address(address)
        if ip.is_private or ip.is_loopback or ip.is_link_local or ip.is_reserved or ip.is_multicast:
            return f"resolves to a non-public address {address}"
    return None


def fetch_text(url, cache):
    """The page's visible text, or a reason it was not fetched. Redirects are
    followed by hand so every hop passes the same checks."""
    if url in cache:
        return cache[url]
    opener = build_opener(_NoRedirect)
    current = url
    result = ("", "too many redirects")
    for _ in range(MAX_HOPS):
        reason = _refused(current)
        if reason:
            result = ("", reason)
            break
        try:
            with opener.open(Request(current, headers={"User-Agent": "trap-test/1"}), timeout=TIMEOUT) as response:
                body = response.read(MAX_BYTES + 1)
            if len(body) > MAX_BYTES:
                result = ("", "page too large")
                break
            parser = _Text()
            parser.feed(body.decode("utf-8", errors="replace"))
            result = (" ".join(parser.parts), None)
            break
        except HTTPError as error:
            if error.code in (301, 302, 303, 307, 308) and error.headers.get("Location"):
                current = urljoin(current, error.headers["Location"])
                continue
            result = ("", f"HTTP {error.code}")
            break
        except (URLError, TimeoutError, OSError) as error:
            result = ("", f"fetch failed: {type(error).__name__}")
            break
    cache[url] = result
    return result


def main(out, runs, model):
    key = VENDORS["openai"].credential_kwargs(os.environ, CredentialMode.API_KEY)["api_key"]
    cache, results = {}, []
    for run in range(1, runs + 1):
        for start in range(0, len(ITEMS), 8):
            batch = ITEMS[start:start + 8]
            lines = "\n".join(f"- id {i}: {svc}: {prop}" for i, svc, prop, _ in batch)
            reply = completion(model=model, api_key=key, response_format={"type": "json_object"},
                               messages=[{"role": "user", "content": PROMPT + lines}])
            answers = json.loads(reply.choices[0].message.content)["answers"]
            for answer in answers:
                text, reason = fetch_text(str(answer.get("source", "")), cache)
                answer["fetch_refused"] = reason
                answer["quote_on_page"] = bool(text) and verify_quote(str(answer.get("quote", "")), text)
            results.append({"run": run, "answers": answers,
                            "usage": {"prompt_tokens": reply.usage.prompt_tokens,
                                      "completion_tokens": reply.usage.completion_tokens}})
        print(f"run {run} done", flush=True)
    Path(out).write_text(json.dumps({"model": model, "items": ITEMS, "results": results}, indent=1))


if __name__ == "__main__":
    main(sys.argv[1], int(sys.argv[2]), sys.argv[3])
```
